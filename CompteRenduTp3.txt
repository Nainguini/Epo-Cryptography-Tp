Etape N°2
	Description de la configuration nginx
	
	server {
			listen 443 ssl;				==> indique le port d'ecoute du canal sécurisé de ssl
			server_name www.tp.local;	==> c'est le nom du serveur, on accede au serveur en utilisant ce nom dans le navigateur
		
			ssl_certificate /etc/nginx/tls/serveur_chaine.crt;	# qle certificat signé par l'AC
			ssl_certificate_key /etc/nginx/tls/serveur.key;		# la clé secrete du serveur qui dechiffrer ce qu'il va recevoir
			
			ssl_protocols TLSv1.2 TLSv1.3;						# le protocol ed securite utilisé
			ssl_prefer_server_ciphers off;
			ssl_session_tickets off;
			
			# Authentification mutuelle des clients
			ssl_client_certificate /etc/nginx/tls/racine.crt;
			ssl_verify_client on;
			ssl_verify_depth 2;
			ssl_crl /etc/nginx/tls/crls.pem;
			
			add_header Strict-Transport-Security "max-age=31536000" always;
			
			location / {
			return 200 "Bonjour $ssl_client_s_dn\n";
			}
		}

Etape N°3
		Observer la poignée de main
		
		
		Avant tout il faut creer une certificat pour bob et la signé
		
		$ openssl genpkey -algorithm EC -pkeyopt ec_paramgen_curve:P-256 -out bob.key
		$ openssl req -new -key bob.key -subj "/CN=alice@tp.local" -out bob.csr
		$ openssl ca -batch -config inter.cnf -extensions ext_client -days 90 -in bob.csr -out bob.crt


	Observer la poigné de main
	
		$ openssl s_client -connect www.tp.local:443 -servername www.tp.local -verify_hostname www.tp.local -CAfile racine/racine.crt -cert bob.crt -key bob.key -cert_chain inter/inter.crt -brief
